cmake_minimum_required(VERSION 3.16)
project(cannge_metadata_tests LANGUAGES CXX)

enable_testing()
if(MSVC)
    add_compile_options(/utf-8)
endif()
add_executable(test-cannge-view test-view.cpp metadata-stubs.cpp ../view.cpp)
target_compile_features(test-cannge-view PRIVATE cxx_std_17)
target_include_directories(test-cannge-view PRIVATE .. ../../../include)
add_test(NAME cannge-view COMMAND test-cannge-view)

add_executable(test-cannge-plan test-plan.cpp metadata-stubs.cpp support-stubs.cpp ../supports.cpp ../plan.cpp ../plan-key.cpp ../view.cpp ../options.cpp)
target_compile_features(test-cannge-plan PRIVATE cxx_std_17)
target_include_directories(test-cannge-plan PRIVATE stubs .. ../.. ../../../include)
add_test(NAME cannge-plan COMMAND test-cannge-plan)

add_executable(test-cannge-supports test-supports.cpp metadata-stubs.cpp support-stubs.cpp ../supports.cpp ../options.cpp)
target_compile_features(test-cannge-supports PRIVATE cxx_std_17)
target_include_directories(test-cannge-supports PRIVATE stubs .. ../.. ../../../include)
add_test(NAME cannge-supports COMMAND test-cannge-supports)

add_executable(test-cannge-lifecycle test-lifecycle.cpp ../context.cpp ../ge.cpp)
target_compile_features(test-cannge-lifecycle PRIVATE cxx_std_17)
target_include_directories(test-cannge-lifecycle PRIVATE stubs .. ../.. ../../../include)
add_test(NAME cannge-lifecycle COMMAND test-cannge-lifecycle)

add_executable(test-cannge-cpy test-cpy.cpp metadata-stubs.cpp ../cpy.cpp ../debug.cpp)
target_compile_features(test-cannge-cpy PRIVATE cxx_std_17)
target_include_directories(test-cannge-cpy PRIVATE stubs .. ../.. ../../../include)
add_test(NAME cannge-cpy COMMAND test-cannge-cpy)

add_executable(test-cannge-build-lookup test-build-lookup.cpp metadata-stubs.cpp support-stubs.cpp ../graph-build.cpp ../graph-build-view.cpp ../graph-build-io.cpp ../tensor.cpp ../mul-mat.cpp ../supports.cpp ../options.cpp ../view.cpp)
target_compile_features(test-cannge-build-lookup PRIVATE cxx_std_17)
target_include_directories(test-cannge-build-lookup PRIVATE stubs .. ../.. ../../../include)
add_test(NAME cannge-build-lookup COMMAND test-cannge-build-lookup)

foreach(test view plan supports lifecycle cpy build-lookup)
    target_sources(test-cannge-${test} PRIVATE abort-stubs.cpp)
endforeach()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(cannge-status-guard SHARED strict-status.cpp)
    target_compile_features(cannge-status-guard PRIVATE cxx_std_17)
    target_include_directories(cannge-status-guard PRIVATE ../../../include)
    target_link_libraries(cannge-status-guard PRIVATE ${CMAKE_DL_LIBS})
